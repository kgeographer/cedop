# Assumes your FastAPI server listens on: http://127.0.0.1:8001

<VirtualHost *:80>
  ServerName cedop.kgeographer.org

  # Logs
  ErrorLog ${APACHE_LOG_DIR}/cedop_error.log
  CustomLog ${APACHE_LOG_DIR}/cedop_access.log combined

  # Proxy settings
  ProxyPreserveHost On

  # Forwarded headers so FastAPI can generate correct URLs and logs
  RequestHeader set X-Forwarded-Proto "http"
  RequestHeader set X-Forwarded-Port "80"

  # If you use websockets (not sure you do), Apache 2.4.47+ can proxy ws with mod_proxy_wstunnel.
  # For now, plain HTTP proxy is enough.

  # Main reverse proxy
  ProxyPass        / http://127.0.0.1:8001/ retry=0 timeout=60
  ProxyPassReverse / http://127.0.0.1:8001/

  # Optional hardening: stop Apache from acting as an open proxy
  <Proxy *>
    Require all granted
  </Proxy>

  # If you serve static files directly via Apache instead of FastAPI, you can add:
  # Alias /static/ /srv/cedop/app/static/
  # <Directory /srv/cedop/app/static/>
  #   Require all granted
  # </Directory>
RewriteEngine on
RewriteCond %{SERVER_NAME} =cedop.kgeographer.org
RewriteRule ^ https://%{SERVER_NAME}%{REQUEST_URI} [END,NE,R=permanent]
</VirtualHost>
